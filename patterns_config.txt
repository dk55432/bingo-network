# ------------------------------------------------------------
# PATTERNS
# Format:
#    PATTERN patternname
#    maskname
#
# or
#
#    PATTERN patternname
#    AtLeast(maskname, min_num_cells_marked)
#
# or
#
#    PATTERN patternname
#    AtMost(maskname, max_num_cells_marked)
#
# or
#
#    PATTERN patternname
#    AND
#        mask1
#        mask2
#        ...etc...
#
# or
#
#    PATTERN patternname
#    OR
#        mask1
#        mask2
#
# Note, you can use AND or OR to combine multiple masks, and AND or OR
#       can be nested, like this:
#
#    PATTERN patternname
#    OR
#        AND
#            col0
#            col4
#        AND
#            row0
#            row4
#
# This would translate into (col0 AND col4) OR (row0 AND row4).
# ------------------------------------------------------------

PATTERN top3-and-bottom3
AND
    AtLeast(row0, 3)
    AtLeast(row4, 3)

PATTERN standard-bingo
